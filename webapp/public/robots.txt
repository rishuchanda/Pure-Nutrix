# Pure Nutrix — crawler directives
#
# Note: a crawler obeys ONLY the most specific user-agent block that matches it.
# The previous file gave Googlebot and Bingbot their own blocks with "Allow: /"
# and no Disallow lines, so those two crawlers ignored the Disallow rules in the
# "*" block entirely and were free to crawl /admin, /cart and /account.
# Every block below therefore repeats the same private paths.

User-agent: *
Allow: /
Disallow: /admin
Disallow: /cart
Disallow: /checkout
Disallow: /account
Disallow: /buy/
Disallow: /order/
Disallow: /whatsapp

User-agent: Googlebot
Allow: /
Disallow: /admin
Disallow: /cart
Disallow: /checkout
Disallow: /account
Disallow: /buy/
Disallow: /order/
Disallow: /whatsapp

User-agent: Bingbot
Allow: /
Disallow: /admin
Disallow: /cart
Disallow: /checkout
Disallow: /account
Disallow: /buy/
Disallow: /order/
Disallow: /whatsapp

User-agent: Googlebot-Image
Allow: /assets/
Allow: /images/

User-agent: Applebot
Allow: /
Disallow: /admin
Disallow: /cart
Disallow: /checkout
Disallow: /account
